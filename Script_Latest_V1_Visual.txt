// ── Сокращённые хелперы ───────────────────────────────────────────────────

// Строгая математическая защита от сингулярности (Division by zero)
// Гарантирует, что знаменатель не попадает в эпсилон-окрестность нуля.
f_safe_div(float num, float den, float eps = 1e-9) =>
    math.abs(den) < eps ? 0.0 : num / den

// Классификация зоны по Score → текстовый тир
f_hint_tier(float score) =>
    score >= 100 ? "BEST" : score >= 80 ? "STRONG" : score >= 40 ? "NORMAL" : "PIVOT"

// V287: Декодирование bit-packed zone meta из ENGINE (Sup1/Sup2/Res1/Res2 Meta каналов)
// Inverse of ENGINE's f_encode_zone_meta(). Returns: [ztype, tier, score, is_prop, is_bull]
// Pine v6 specifics: оператор '/' на int возвращает float — поэтому используем math.floor() для целочисленного деления
f_decode_zone_meta(float meta) =>
    int raw     = int(math.round(math.max(meta, 0.0)))
    int z_code  = int(math.floor(float(raw)         )) % 16
    int t_code  = int(math.floor(float(raw) / 16.0  )) % 16
    int s_code  = int(math.floor(float(raw) / 256.0 )) % 128
    int p_int   = int(math.floor(float(raw) / 32768.0)) % 2
    int b_int   = int(math.floor(float(raw) / 65536.0)) % 2
    bool p_flag = p_int == 1
    bool b_flag = b_int == 1
    float score_decoded = float(s_code) / 127.0 * 150.0
    string ztype = "MITIG"
    if z_code == 0
        ztype := "OB"
    else if z_code == 1
        ztype := "BB"
    else if z_code == 2
        ztype := "RZ"
    else if z_code == 3
        ztype := "IMB"
    else if z_code == 4
        ztype := "ACC"
    else if z_code == 5
        ztype := "DST"
    else if z_code == 6
        ztype := "VCP"
    else if z_code == 7
        ztype := "MBR"
    else if z_code == 8
        ztype := "PROP"
    string tier = "BEST"
    if t_code == 0
        tier := "PIVOT"
    else if t_code == 1
        tier := "NORMAL"
    else if t_code == 2
        tier := "STRONG"
    [ztype, tier, score_decoded, p_flag, b_flag]

// Pine v6: используем if/else с реассайном (более безопасно чем глубокая тернарная цепь)
f_zone_color(string ztype, bool is_bull) =>
    color base = color.new(#00C853, 0)
    if ztype == "OB"
        base := is_bull ? color.new(#00C853, 0) : color.new(#FF1744, 0)
    else if ztype == "BB"
        base := is_bull ? color.new(#00E5FF, 0) : color.new(#FF6B6B, 0)
    else if ztype == "RZ"
        base := is_bull ? color.new(#FFEB3B, 0) : color.new(#FF9800, 0)
    else if ztype == "MBR"
        base := is_bull ? color.new(#CDDC39, 0) : color.new(#FF5722, 0)
    else if ztype == "IMB"
        base := is_bull ? color.new(#FFD700, 0) : color.new(#FFA000, 0)
    else if ztype == "ACC"
        base := color.new(#7C4DFF, 0)
    else if ztype == "DST"
        base := color.new(#9575CD, 0)
    else if ztype == "VCP"
        base := color.new(#90A4AE, 0)
    else if ztype == "PROP"
        base := is_bull ? color.new(#00E676, 0) : color.new(#FF1744, 0)
    else if ztype == "MITIG"
        base := is_bull ? color.new(#26A69A, 0) : color.new(#EF5350, 0)
    base

// V287: Толщина рамки от типа зоны (OB и BB — толстые, остальные — тонкие)
f_zone_border_width(string ztype) =>
    ztype == "OB" or ztype == "BB" or ztype == "PROP" or ztype == "MITIG" ? 2 : 1

// V287: Транспарентность из decoded score (плавный градиент score → transparency)
// V287.1: Параметр theme управляет диапазоном:
//   "Light" / "Auto"  → 95→40 (исторически безопасный диапазон, сохранён по
//                       умолчанию для обратной совместимости с существующими
//                       пользовательскими настройками — поведение не меняется,
//                       пока пользователь явно не переключится на "Dark");
//   "Dark"            → 80→20 (зоны заметнее на тёмном фоне).
f_score_to_transp_v(float score, string theme) =>
    float s_clamped = math.min(math.max(score, 0.0), 150.0)
    int t_max = 95
    int t_min = 40
    if theme == "Dark"
        t_max := 80
        t_min := 20
    int span = t_max - t_min
    int t = int(math.round(t_max - s_clamped * span / 150.0))
    math.max(t_min, math.min(t_max, t))

// V287.1: Динамический glow-offset — гарантирует видимость glow на узких зонах.
//         Берём 5% высоты зоны, но не меньше 4× minimum tick инструмента.
f_glow_pad(float top, float bot) =>
    math.max(syminfo.mintick * 4.0, (top - bot) * 0.05)

// V287.2: Helper для sparkline-блока (Trend Strength meter).
// Возвращает Unicode-символ блока в зависимости от индекса позиции (1..8) —
// первые 2 позиции тонкие, средние нормальные, последние толстые.
f_spark_block(int idx) =>
    string ch = "▂"
    if idx > 6
        ch := "█"
    else if idx > 4
        ch := "▆"
    else if idx > 2
        ch := "▄"
    ch

int  RM_MIN_FOR_RISK = 5
int  RM_MIN_FOR_MC   = 10
int  RM_MC_MAX_STEPS = 50
var array<float> _v_trade_returns = array.new<float>()
if v_total_tr > 0 and v_total_tr != nz(v_total_tr[1], 0)
    float _v_last_pnl = (v_gross_profit - nz(v_gross_profit[1], 0.0)) - (v_gross_loss - nz(v_gross_loss[1], 0.0))
    float _v_last_eq  = math.max(v_equity, 1.0)

    array.push(_v_trade_returns, _v_ret)
    if array.size(_v_trade_returns) > RM_MAX_RETURNS

// V287.4 (perf-fix): Sharpe/Sortino/MC использовались ТОЛЬКО в дашборде
// (рендерится на barstate.islast) и в alertcondition (стреляет только на
// realtime-баре). Раньше пересчитывались на каждом историческом баре, что
// при подключённом ENGINE и накопленных трейдах давало ~125M операций
// (500 MC × до 50 шагов × ~5000 баров истории) и приводило к ошибке
// TradingView «indicator timed out» (>20с) при подключении VISUAL к ENGINE.
// Теперь это `var float` и обновляются только на последнем баре —
// подключение и первичная отрисовка ускоряются на несколько порядков.
var float v_sharpe  = 0.0
var float v_sortino = 0.0
var float v_mc_dd95 = 0.0

    if _v_ret_n >= RM_MIN_FOR_RISK
        float _v_ret_sum = 0.0
        for _ri = 0 to _v_ret_n - 1
            _v_ret_sum += array.get(_v_trade_returns, _ri)
        float _v_ret_mean = f_safe_div(_v_ret_sum, float(_v_ret_n))
        float _v_ret_sq_sum = 0.0
        float _v_ret_down_sq = 0.0
        for _ri = 0 to _v_ret_n - 1
            float _r = array.get(_v_trade_returns, _ri)
            _v_ret_sq_sum += math.pow(_r - _v_ret_mean, 2)
            if _r < 0
                _v_ret_down_sq += math.pow(_r, 2)
        float _v_ret_std = math.sqrt(f_safe_div(_v_ret_sq_sum, float(math.max(_v_ret_n - 1, 1))))
        float _v_down_std = math.sqrt(f_safe_div(_v_ret_down_sq, float(math.max(_v_ret_n - 1, 1))))
        v_sharpe  := f_safe_div(_v_ret_mean, _v_ret_std) * math.sqrt(252.0)
        v_sortino := f_safe_div(_v_ret_mean, _v_down_std) * math.sqrt(252.0)
    else
        v_sharpe  := 0.0
        v_sortino := 0.0

    // V286: Monte Carlo Simulation (500 iterations, block bootstrap)
    if _v_ret_n >= RM_MIN_FOR_MC
        var array<float> _v_mc_dds = array.new<float>()
        array.clear(_v_mc_dds)
        for _sim = 0 to 499
            float _mc_eq = 1.0
            float _mc_peak = 1.0
            float _mc_max_dd = 0.0
            int _block_size = 3 + (_sim % 3)
            for _step = 0 to math.min(_v_ret_n - 1, RM_MC_MAX_STEPS - 1)
                int _block_start = (_sim * 7 + (_step / _block_size) * 13) % _v_ret_n
                int _pick = (_block_start + (_step % _block_size)) % _v_ret_n
                float _r = array.get(_v_trade_returns, _pick)
                _mc_eq := _mc_eq * (1.0 + _r)
                _mc_peak := math.max(_mc_peak, _mc_eq)
                float _dd = f_safe_div(_mc_peak - _mc_eq, _mc_peak)
                _mc_max_dd := math.max(_mc_max_dd, _dd)
            array.push(_v_mc_dds, _mc_max_dd)
        array.sort(_v_mc_dds)
        int _p95_idx = math.min(474, array.size(_v_mc_dds) - 1)
        v_mc_dd95 := array.get(_v_mc_dds, _p95_idx) * 100.0
    else
        v_mc_dd95 := 0.0

    // ── ZONE_DETECTED → ZONE_TOUCH | back to WAIT if invalid ──
    if hint_state == HS_ZONE_DET
        if not _zone_valid
            hint_state := HS_WAIT
        else if _touched_long
            hint_state     := HS_ZONE_TOUCH
            hint_zone_side := "support"
            hint_zone_type := _top1_lookup_type
        else if _touched_short
            hint_state     := HS_ZONE_TOUCH
            hint_zone_side := "resistance"
            hint_zone_type := _top1_lookup_type

    // ── ZONE_TOUCH → ENTRY_LONG | ENTRY_SHORT | back to WAIT after timeout ──
    if hint_state == HS_ZONE_TOUCH
        if hint_zone_side == "support" and _confirm_long
            // Entry LONG
            hint_state         := HS_ENTRY_L
            hint_entry_long    := true
            hint_entry_price   := v_sim_entry > 0 ? v_sim_entry : close
            hint_sl_price      := v_sim_sl  > 0 ? v_sim_sl  : (v_top1_price - 0.5 * nz(atr_val, 1.0))
            hint_orig_sl       := hint_sl_price
            hint_tp1_price     := v_sim_tp1 > 0 ? v_sim_tp1 : (hint_entry_price + 1.5 * (hint_entry_price - hint_sl_price))

            hint_sl_price      := v_sim_sl  > 0 ? v_sim_sl  : (v_top1_price + 0.5 * nz(atr_val, 1.0))

    var float s_low    = na

            box.set_bottom(s_box, s_low)
            box.set_right(s_box, bar_index + 1)
    if session_end and not na(s_box)
        box.set_right(s_box, bar_index)
    in_sess
